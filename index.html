import math import json import os from datetime import datetime from fastapi import FastAPI, HTTPException from pydantic import BaseModel app = FastAPI(title="Advanced Football Analytics Engine") # File Logging Directory Setup LOG_DIR = "match_logs" if not os.path.exists(LOG_DIR): os.makedirs(LOG_DIR) class MatchDataInput(BaseModel): match_id: str home_team: str away_team: str # Corner Metrics home_avg_corners: float away_avg_corners: float # Yellow Cards Metrics (Last 5 Games) home_cards_last5: list[int] # e.g., [2, 1, 3, 0, 2] away_cards_last5: list[int] # e.g., [3, 2, 4, 1, 2] h2h_avg_cards: float # တွေ့ဆုံမှုများမှ ပျမ်းမျှ ကတ်အရေအတွက် # Possession Metrics home_avg_possession: float # Percentage (e.g., 58.5) away_avg_possession: float # Market Odds Metrics market_555_home: float # ပြင်ပေါက်ကြေး (555 Mix Odds) market_555_draw: float market_555_away: float xbet_home: float # 1xBet Odds xbet_draw: float xbet_away: float @app.post("/api/advanced-predict") def analyze_and_predict(data: MatchDataInput): # 1. Corner Prediction Logic total_expected_corners = data.home_avg_corners + data.away_avg_corners more_corners = data.home_team if data.home_avg_corners > data.away_avg_corners else data.away_team corner_diff = abs(data.home_avg_corners - data.away_avg_corners) # 2. Yellow Cards Logic (Last 5 Avg + H2H Avg) home_cards_avg = sum(data.home_cards_last5) / len(data.home_cards_last5) away_cards_avg = sum(data.away_cards_last5) / len(data.away_cards_last5) predicted_total_cards = round((home_cards_avg + away_cards_avg + data.h2h_avg_cards) / 2, 1) # 3. Possession Prediction Logic possession_total = data.home_avg_possession + data.away_avg_possession predicted_home_possession = round((data.home_avg_possession / possession_total) * 100, 1) predicted_away_possession = round(100 - predicted_home_possession, 1) more_possession_team = data.home_team if predicted_home_possession > predicted_away_possession else data.away_team # 4. 555 Mix vs 1xBet Odds Value Analysis # Calculated implied probability gap between 555 Mix and 1xBet home_value_gap = round(((1 / data.xbet_home) - (1 / data.market_555_home)) * 100, 2) away_value_gap = round(((1 / data.xbet_away) - (1 / data.market_555_away)) * 100, 2) value_bet_recommendation = "NEUTRAL" if home_value_gap > 3.0: value_bet_recommendation = f"VALUE ON {data.home_team} (1xBet offers better price than 555)" elif away_value_gap > 3.0: value_bet_recommendation = f"VALUE ON {data.away_team} (1xBet offers better price than 555)" # Consolidated Results Sheet result_payload = { "match_id": data.match_id, "timestamp": datetime.utcnow().isoformat(), "teams": {"home": data.home_team, "away": data.away_team}, "corner_analysis": { "more_corners_team": more_corners, "expected_total_corners": round(total_expected_corners, 1), "corner_margin": round(corner_diff, 1) }, "cards_analysis": { "home_last5_card_avg": round(home_cards_avg, 2), "away_last5_card_avg": round(away_cards_avg, 2), "h2h_card_avg": data.h2h_avg_cards, "predicted_total_cards": predicted_total_cards }, "possession_analysis": { "more_possession_team": more_possession_team, "predicted_possession_pct": { "home": predicted_home_possession, "away": predicted_away_possession } }, "odds_comparison": { "555_mix": {"home": data.market_555_home, "draw": data.market_555_draw, "away": data.market_555_away}, "1xbet": {"home": data.xbet_home, "draw": data.xbet_draw, "away": data.xbet_away}, "value_bet_finding": value_bet_recommendation } } # 5. File Logging System - Match Data & Analytics saving per Match ID file_path = os.path.join(LOG_DIR, f"{data.match_id}_{data.home_team}_vs_{data.away_team}.json") with open(file_path, "w", encoding="utf-8") as f: json.dump(result_payload, f, ensure_ascii=False, indent=4) return { "status": "SUCCESS", "file_logged": file_path, "analytics": result_payload }
